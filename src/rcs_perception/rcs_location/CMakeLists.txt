cmake_minimum_required(VERSION 3.16)
project(rcs_location LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/RcsLocationState.msg")
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(rcs_location_algorithm_imu STATIC algorithm/src/rcs_location_algorithm_imu.cpp)
target_include_directories(rcs_location_algorithm_imu PUBLIC algorithm/include)
ament_target_dependencies(rcs_location_algorithm_imu tf2)
add_executable(rcs_location_algorithm_node algorithm/src/rcs_location_algorithm_node.cpp)
rosidl_get_typesupport_target(state_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(rcs_location_algorithm_node rcs_location_algorithm_imu "${state_typesupport}")
ament_target_dependencies(rcs_location_algorithm_node rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs tf2_ros)
add_executable(rcs_location_algorithm_test test/algorithm/test_imu.cpp)
target_link_libraries(rcs_location_algorithm_test rcs_location_algorithm_imu)
enable_testing()
add_test(NAME imu_algorithm COMMAND rcs_location_algorithm_test)
install(TARGETS rcs_location_algorithm_node rcs_location_algorithm_test DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/rcs_location scripts/test DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch test DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(CODE "
  file(MAKE_DIRECTORY \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/share/${PROJECT_NAME}/config\")
  foreach(name rcs_location rcs_location_algorithm)
    file(REMOVE \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/share/${PROJECT_NAME}/config/\${name}.yaml\")
    file(CREATE_LINK \"${CMAKE_CURRENT_SOURCE_DIR}/config/\${name}.yaml\"
      \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/share/${PROJECT_NAME}/config/\${name}.yaml\" SYMBOLIC)
  endforeach()
")
ament_export_dependencies(rosidl_default_runtime)
ament_package()
