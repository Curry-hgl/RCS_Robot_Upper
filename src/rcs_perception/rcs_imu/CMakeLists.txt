cmake_minimum_required(VERSION 3.16)
project(rcs_imu LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
add_subdirectory(third_party/lpms)
add_executable(rcs_imu_node src/rcs_imu_node.cpp)
target_include_directories(rcs_imu_node PRIVATE include)
target_link_libraries(rcs_imu_node rcs_imu_lpms)
ament_target_dependencies(rcs_imu_node rclcpp sensor_msgs geometry_msgs std_msgs std_srvs tf2)
add_executable(rcs_imu_conversion_test test/test_conversion.cpp)
target_include_directories(rcs_imu_conversion_test PRIVATE include)
target_link_libraries(rcs_imu_conversion_test rcs_imu_lpms)
ament_target_dependencies(rcs_imu_conversion_test sensor_msgs tf2)
install(TARGETS rcs_imu_node rcs_imu_conversion_test DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/rcs_imu scripts/test DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch test DESTINATION share/${PROJECT_NAME}
    PATTERN "__pycache__" EXCLUDE)
install(FILES third_party/lpms/LICENSE.txt DESTINATION share/${PROJECT_NAME}/third_party/lpms)
# 开发配置直接链接源码，普通 colcon build 后也可修改。
install(CODE "
    file(MAKE_DIRECTORY \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/share/${PROJECT_NAME}/config\")
    file(REMOVE \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/share/${PROJECT_NAME}/config/rcs_imu.yaml\")
    file(CREATE_LINK \"${CMAKE_CURRENT_SOURCE_DIR}/config/rcs_imu.yaml\"
        \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/share/${PROJECT_NAME}/config/rcs_imu.yaml\" SYMBOLIC)
")
ament_package()
